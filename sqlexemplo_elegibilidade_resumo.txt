/*
================================================================================
SQL ELEGIBILIDADE RESUMO — 1 linha por colaborador (mês / período)
Statistical Institute of Jamaica
================================================================================

PURPOSE (different from sqlexemplo_power / sqlexemplo_resumo):
  Jamaica daily rule: vacation accrues ONLY on days the employee is active
  (= not on a blocking leave). ANY blocking leave day → no vacation accrual
  that day — even if the absence is shorter than 14 calendar days.

  This report answers:
    • How many calendar days in the period were "off" (blocking leave)?
    • Jamaica accrual for the period = Eligible_days × (tier / 365|366)
    • Gap ≈ days_off × rate  (= vacation Factorial may still have accrued
      while the person was off — for manual HR adjustment)

  The >14 "extended pause" flag is EXTRA process info (month-end pack,
  future balance narrative). It does NOT mean short leave still accrues.

COMPARE TO:
  • sqlexemplo_power.txt / sqlexemplo_resumo.txt
      → gap ONLY when LOA >14 calendar days (pause block)
  • THIS FILE
      → gap for EVERY blocking-leave day in the period (Ex. 1 Bernarda)

Filters: {{Data_init}} / {{Data_end}}
Demo tips:
  • Example 1 Bernarda Baker (sick Jun 1–7)
      Data_init=2026-06-01  Data_end=2026-06-30
      Expect: Days_Off≈7  Eligible≈23  Jamaica≈1.2603  Gap≈0.3836
  • Example 2 Diana Davis (suspension Aug 10–20)
      Data_init=2026-08-01  Data_end=2026-08-31
      Expect: Days_Off≈11 Eligible≈20  Jamaica≈1.0959  Gap≈0.6027
  • Example 3 Charles Carter (vacation Dec 1–18)
      Data_init=2026-12-01  Data_end=2026-12-31
      Expect: Days_Off≈18 Eligible≈13 Jamaica≈0.7123 Gap≈0.9863
        Has_Extended_Pause=YES

Honest limit:
  Factorial does NOT stop vacation accrual on leave days. Gap = manual adjust.
  Overlapping leaves in the same days are summed then CAPPED at period length.
================================================================================
*/

WITH params AS (
    SELECT
        14 AS pause_threshold_days,
        CAST('{{Data_init}}' AS date) AS period_start,
        CAST('{{Data_end}}'  AS date) AS period_end,
        date_diff(
            'day',
            CAST('{{Data_init}}' AS date),
            CAST('{{Data_end}}' AS date)
        ) + 1 AS days_in_period,
        CASE
            WHEN MOD(year(CAST('{{Data_init}}' AS date)), 400) = 0 THEN 366
            WHEN MOD(year(CAST('{{Data_init}}' AS date)), 100) = 0 THEN 365
            WHEN MOD(year(CAST('{{Data_init}}' AS date)), 4)   = 0 THEN 366
            ELSE 365
        END AS days_in_year
),

employee_base AS (
    SELECT
        e.id AS employee_id,
        e.company_identifier AS employee_code,
        e.full_name AS employee_name,
        CAST(e.tenure_start_date AS date) AS tenure_start_date,
        ser.annual_vacation_days,
        ser.max_accumulation_days,
        p.days_in_year,
        p.days_in_period,
        p.period_start,
        p.period_end,
        p.pause_threshold_days,
        CASE
            WHEN ser.annual_vacation_days IS NOT NULL
            THEN ser.annual_vacation_days / CAST(p.days_in_year AS DOUBLE)
            ELSE NULL
        END AS accrual_per_calendar_day
    FROM employees e
    CROSS JOIN params p
    INNER JOIN sij_employee_rates ser ON ser.employee_id = e.id
),

blocking_leave_slices AS (
    SELECT
        eb.employee_id,
        l.id AS leave_id,
        l.leave_type_name AS absence_type,
        CAST(l.start_on AS date) AS start_date,
        CAST(l.finish_on AS date) AS end_date,
        date_diff('day', CAST(l.start_on AS date), CAST(l.finish_on AS date)) + 1
            AS calendar_days_full,
        -- Clip leave to the report period
        date_diff(
            'day',
            CASE
                WHEN CAST(l.start_on AS date) > eb.period_start
                THEN CAST(l.start_on AS date)
                ELSE eb.period_start
            END,
            CASE
                WHEN CAST(l.finish_on AS date) < eb.period_end
                THEN CAST(l.finish_on AS date)
                ELSE eb.period_end
            END
        ) + 1 AS days_off_in_period,
        CASE
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%holiday%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%vacation%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%annual%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%férias%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%ferias%'
            THEN 'vacation'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%sick%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%illness%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%medical%'
            THEN 'sick'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%suspend%'
            THEN 'suspension'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%matern%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%patern%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%parental%'
            THEN 'maternity'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%study%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%training%'
            THEN 'study'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%compassionate%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%bereavement%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%grievance%'
            THEN 'compassionate'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%special%'
            THEN 'special'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%department%'
            THEN 'departmental'
            WHEN LOWER(COALESCE(l.leave_type_name, '')) LIKE '%unpaid%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%no pay%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%nopay%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%leave of absence%'
              OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%loa%'
            THEN 'other_loa'
            ELSE 'other_absence'
        END AS leave_family,
        CASE
            WHEN date_diff('day', CAST(l.start_on AS date), CAST(l.finish_on AS date)) + 1
                 > eb.pause_threshold_days
            THEN 1
            ELSE 0
        END AS is_extended_pause_absence
    FROM leaves l
    JOIN employee_base eb ON eb.employee_id = l.employee_id
    WHERE LOWER(l.status) IN ('approved', 'accepted')
      AND l.start_on IS NOT NULL
      AND l.finish_on IS NOT NULL
      AND CAST(l.start_on AS date) <= eb.period_end
      AND CAST(l.finish_on AS date) >= eb.period_start
      AND LOWER(COALESCE(l.leave_type_name, '')) NOT LIKE '%working from home%'
      AND LOWER(COALESCE(l.leave_type_name, '')) NOT LIKE '%work from home%'
      AND LOWER(COALESCE(l.leave_type_name, '')) NOT LIKE '%wfh%'
      -- Blocking families = Jamaica "not actively working" for vacation accrual
      AND (
            LOWER(COALESCE(l.leave_type_name, '')) LIKE '%holiday%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%vacation%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%annual%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%férias%'

         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%suspend%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%matern%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%patern%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%parental%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%study%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%training%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%compassionate%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%bereavement%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%grievance%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%special%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%department%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%unpaid%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%no pay%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%nopay%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%leave of absence%'
         OR LOWER(COALESCE(l.leave_type_name, '')) LIKE '%loa%'
      )
),

per_employee AS (
    SELECT
        eb.employee_id,
        eb.employee_code,
        eb.employee_name,
        eb.tenure_start_date,
        eb.annual_vacation_days,
        eb.max_accumulation_days,
        eb.days_in_year,
        eb.days_in_period,
        eb.period_start,
        eb.period_end,
        eb.pause_threshold_days,
        eb.accrual_per_calendar_day,
        COALESCE(SUM(s.days_off_in_period), 0) AS days_off_raw_sum,
        COALESCE(COUNT(s.leave_id), 0) AS blocking_absences_count,
        COALESCE(MAX(s.is_extended_pause_absence), 0) AS has_extended_pause_flag
    FROM employee_base eb
    LEFT JOIN blocking_leave_slices s ON s.employee_id = eb.employee_id
    GROUP BY
        eb.employee_id,
        eb.employee_code,
        eb.employee_name,
        eb.tenure_start_date,
        eb.annual_vacation_days,
        eb.max_accumulation_days,
        eb.days_in_year,
        eb.days_in_period,
        eb.period_start,
        eb.period_end,
        eb.pause_threshold_days,
        eb.accrual_per_calendar_day
),

calc AS (
    SELECT
        pe.*,
        -- Cap: overlapping leaves cannot exceed period length
        CASE
            WHEN pe.days_off_raw_sum > pe.days_in_period THEN pe.days_in_period
            ELSE pe.days_off_raw_sum
        END AS days_on_blocking_leave,
        pe.days_in_period - CASE
            WHEN pe.days_off_raw_sum > pe.days_in_period THEN pe.days_in_period
            ELSE pe.days_off_raw_sum
        END AS eligible_days,
        ROUND(pe.days_in_period * pe.accrual_per_calendar_day, 4)
            AS factorial_accrual_if_full_period,
        ROUND(
            (pe.days_in_period - CASE
                WHEN pe.days_off_raw_sum > pe.days_in_period THEN pe.days_in_period
                ELSE pe.days_off_raw_sum
            END) * pe.accrual_per_calendar_day
        , 4) AS jamaica_accrual_period,
        ROUND(
            (CASE
                WHEN pe.days_off_raw_sum > pe.days_in_period THEN pe.days_in_period
                ELSE pe.days_off_raw_sum
            END) * pe.accrual_per_calendar_day
        , 4) AS gap_manual_adjustment,
        FLOOR(
            ROUND(
                (CASE
                    WHEN pe.days_off_raw_sum > pe.days_in_period THEN pe.days_in_period
                    ELSE pe.days_off_raw_sum
                END) * pe.accrual_per_calendar_day
            , 6) + 0.5
        ) AS gap_rounded_jamaica_half_up
    FROM per_employee pe
)

SELECT
    employee_code                         AS "Employee_Code",
    employee_name                         AS "Employee_Name",
    tenure_start_date                     AS "Tenure_Start_Date",
    annual_vacation_days                  AS "Annual_Entitlement_Days",
    max_accumulation_days                 AS "Max_Accumulation_3y",
    days_in_year                          AS "Days_In_Year_365_or_366",
    period_start                          AS "Period_Start",
    period_end                            AS "Period_End",
    days_in_period                        AS "Days_In_Period",
    ROUND(accrual_per_calendar_day, 6)    AS "Accrual_Per_Calendar_Day",
    days_on_blocking_leave                AS "Days_On_Blocking_Leave",
    eligible_days                         AS "Eligible_Days_Jamaica",
    blocking_absences_count               AS "Blocking_Absences_Count",
    factorial_accrual_if_full_period      AS "A_Factorial_Accrual_If_Full_Period",
    jamaica_accrual_period                AS "B_Jamaica_Accrual_Period",
    gap_manual_adjustment                 AS "C_Gap_Manual_Adjustment",
    gap_rounded_jamaica_half_up           AS "C_Gap_Rounded_Half_Up",
    CASE
        WHEN has_extended_pause_flag = 1
        THEN 'YES (>14 calendar-day absence in/overlapping period)'
        ELSE 'NO'
    END                                   AS "Has_Extended_Pause_Flag",
    CASE
        WHEN days_on_blocking_leave = 0 THEN
            'No blocking leave in period. Jamaica accrual = full period.'
        WHEN has_extended_pause_flag = 1 THEN
            CONCAT(
                'Off ', CAST(days_on_blocking_leave AS varchar),
                ' day(s). Jamaica accrues ', CAST(eligible_days AS varchar),
                ' eligible day(s) ≈ ', CAST(jamaica_accrual_period AS varchar),
                '. Gap ≈ ', CAST(gap_manual_adjustment AS varchar),
                ' (Factorial may still accrue on off days). Extended pause >14 also applies.'
            )
        ELSE
            CONCAT(
                'Off ', CAST(days_on_blocking_leave AS varchar),
                ' day(s) — accrual should STOP on those days even if ≤14. ',
                'Jamaica period accrual ≈ ', CAST(jamaica_accrual_period AS varchar),
                ' (', CAST(eligible_days AS varchar), ' × rate). ',
                'Gap to adjust ≈ ', CAST(gap_manual_adjustment AS varchar),
                '. Has_Extended_Pause=NO (short leave; still a gap).'
            )
    END                                   AS "Conclusion_Eligibility"
FROM calc
WHERE annual_vacation_days IS NOT NULL
ORDER BY
    CASE WHEN gap_manual_adjustment > 0 THEN 0 ELSE 1 END,
    gap_manual_adjustment DESC,
    employee_name
;
