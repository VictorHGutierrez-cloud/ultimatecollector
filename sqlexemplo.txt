/*
================================================================================
(V1) Jamaica Vacation Accrual — Absence Proof (A/B/C)
File: sqlexemplo.txt (same as sqlexemplo_power.txt)
Statistical Institute of Jamaica
FIX: Insights leaves has NO leave_type_id / leave_types table.
  leave_type_name is often NULL — classify using name, name_slug, OR description
  (SIJ Suspension text in description still matches %suspend%).
================================================================================
REPORT ORDER FOR CLIENT:
  1. (V1) Jamaica Vacation Accrual — Monthly Close Summary   (primary month-end)
  2. (V1) Jamaica Vacation Accrual — Leave Audit Detail
  3. (V1) Jamaica Vacation Accrual — Employee Gap Totals     (optional)
  4. (V1) Jamaica Vacation Accrual — Absence Proof (A/B/C)   ← THIS FILE (demo/proof)

Uses Factorial fields as-is (do NOT reinvent product rules):
  • tenure_start_date     — Factorial tenure start
  • antiquity             — Factorial tenure/antiquity value
  • contract_version_*    — Factorial contract dates (shown only; not recomputed)
  • leaves.status / dates / duration — Factorial absence data

Jamaica rule (Business Case — Daily Leave Accrual):
  • Accrue vacation ONLY on days the employee is active (= not on blocking leave)
  • ANY blocking leave day → Jamaica accrual that day = 0 (Ex.1 sick 7d, Ex.2
    suspension 11d). Length does NOT matter for the daily stop.
  • >14 consecutive calendar days = EXTRA process flag (long vacation / ESL pack),
    NOT a gate for whether HR must adjust.
  • days_in_year 365/366; rounding >= 0.50 up; SIJ tier 15/20/21/25 + 3y cap

Columns A / B / C (per absence):
  • A = what Factorial still accrues during the absence (fixed balance keeps going)
  • B = Jamaica accrual during that absence (= 0 if blocking leave family)
  • C = gap to adjust manually (= A when blocking; 0 only if type is non-blocking)
  • What_Client_Should_Do = plain-language HR action (subtract C from vacation)

Honest limit:
  Factorial has NO native stop_accrual on leaves. C is ALWAYS the HR adjustment
  for blocking absences — 7 days or 70 days, same math (days × rate).

Also see:
  • Employee Gap Totals → sqlexemplo_resumo.txt
  • Monthly Close Summary / Leave Audit Detail → sqlexemplo_elegibilidade_*.txt

Filters: {{Data_init}} / {{Data_end}}
Demo tips (sandbox STATIN):
  • Ex.1 Bernarda Baker sick Jun 1–7 → 2026-06-01 .. 2026-06-30
      Expect C ≈ 7 × (20/365) ≈ 0.3836  (Above_Threshold=NO, but C > 0)
  • Ex.2 Diana Davis suspension Aug → 2026-08-01 .. 2026-08-31
  • Ex.3 Charles Carter vacation Dec 1–18 → 2026-12-01 .. 2026-12-31
      Above_Threshold=YES + C ≈ 18 × rate
  • Clara Cooper Ex.4 (2-day retro) → C still > 0; HR may reverse (doc Ex.4)
================================================================================
*/

employee_base AS (
    SELECT
        e.id AS employee_id,
        e.company_identifier AS employee_code,
        e.full_name AS employee_name,
        -- Factorial native tenure fields (do not recompute "years of service")
        CAST(e.tenure_start_date AS date) AS tenure_start_date,
        e.antiquity AS antiquity,
        CAST(e.contract_version_starts_on AS date) AS contract_version_starts_on,
        CAST(e.contract_version_effective_on AS date) AS contract_version_effective_on,
        ser.annual_vacation_days,
        ser.max_accumulation_days,
        p.days_in_year,
        p.year_start,
        p.year_end,
        p.period_start,
        p.period_end,
        p.pause_threshold_days,
        CASE
            WHEN ser.annual_vacation_days IS NOT NULL
            THEN ser.annual_vacation_days / CAST(p.days_in_year AS DOUBLE)
            ELSE NULL
        END AS accrual_per_calendar_day,
        -- Jamaica prorata input: eligible calendar days from Factorial tenure_start
        CASE
            WHEN CAST(e.tenure_start_date AS date) IS NULL THEN NULL
            WHEN CAST(e.tenure_start_date AS date) > p.year_end THEN 0
            ELSE
                date_diff(
                    'day',
                    CASE
                        WHEN CAST(e.tenure_start_date AS date) > p.year_start
                        THEN CAST(e.tenure_start_date AS date)
                        ELSE p.year_start
                    END,
                    LEAST(p.year_end, p.period_end)
                ) + 1
        END AS eligible_calendar_days_in_year_to_period_end
    FROM employees e
    CROSS JOIN params p
    LEFT JOIN sij_employee_rates ser ON ser.employee_id = e.id
),

absences AS (
    SELECT
        eb.employee_id,
        eb.employee_code,
        eb.employee_name,
        eb.tenure_start_date,
        eb.antiquity,
        eb.contract_version_starts_on,
        eb.contract_version_effective_on,
        eb.annual_vacation_days,
        eb.max_accumulation_days,
        eb.eligible_calendar_days_in_year_to_period_end,
        eb.days_in_year,
        eb.accrual_per_calendar_day,
        eb.pause_threshold_days,
        l.id AS leave_id,
        COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), '(see description)') AS absence_type,
        l.status AS status,
        CAST(l.start_on AS date) AS start_date,
        CAST(l.finish_on AS date) AS end_date,
        date_diff('day', CAST(l.start_on AS date), CAST(l.finish_on AS date)) + 1
            AS calendar_days,
        CASE
            WHEN LOWER(COALESCE(l.duration_type, 'days')) IN ('days', 'day')
            THEN COALESCE(l.duration, 0)
            WHEN LOWER(l.duration_type) IN ('hours', 'hour')
            THEN COALESCE(l.duration, 0) / 8.0
            ELSE COALESCE(l.duration, 0)
        END AS duration_days_factorial,
        date_add('day', 1, CAST(l.finish_on AS date)) AS return_date,
        l.description AS description,
        CASE
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%holiday%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%vacation%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%annual%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%férias%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%ferias%'
            THEN 'vacation'
            -- SIJ Extended Sick Leave + SIJ Sick Leave (both contain 'sick')
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%sick%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%illness%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%medical%'
            THEN 'sick'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%suspend%'
            THEN 'suspension'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%matern%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%patern%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%parental%'
            THEN 'maternity'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%study%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%training%'
            THEN 'study'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%compassionate%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%bereavement%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%grievance%'
            THEN 'compassionate'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%special%'
            THEN 'special'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%department%'
            THEN 'departmental'
            WHEN LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%unpaid%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%no pay%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%nopay%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%leave of absence%'
              OR LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) LIKE '%loa%'
            THEN 'other_loa'
            ELSE 'other_absence'
        END AS leave_family
    FROM leaves l
    JOIN employee_base eb ON eb.employee_id = l.employee_id
    CROSS JOIN params p
    WHERE LOWER(l.status) IN ('approved', 'accepted')
      AND l.start_on IS NOT NULL
      AND l.finish_on IS NOT NULL
      AND CAST(l.start_on AS date) <= p.period_end
      AND CAST(l.finish_on AS date) >= p.period_start
      AND LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) NOT LIKE '%working from home%'
      AND LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) NOT LIKE '%work from home%'
      AND LOWER(COALESCE(NULLIF(TRIM(l.leave_type_name), ''), NULLIF(TRIM(l.leave_type_name_slug), ''), NULLIF(TRIM(l.description), ''), '')) NOT LIKE '%wfh%'
),

counts AS (
    SELECT
        employee_id,
        COUNT(*) AS absences_in_period
    FROM absences
    GROUP BY employee_id
),

with_previous AS (
    SELECT
        a.*,
        c.absences_in_period,
        LAG(a.return_date) OVER (
            PARTITION BY a.employee_id
            ORDER BY a.start_date, a.leave_id
        ) AS previous_return_date
    FROM absences a
    JOIN counts c ON c.employee_id = a.employee_id
),

calc AS (
    SELECT
        wp.*,

        -- Daily rule: blocking leave family → no Jamaica vacation accrual that day
        -- (length irrelevant). Matches Business Case Ex.1 / Ex.2 / opening definition.
        CASE
            WHEN wp.leave_family IN (
                    'vacation', 'sick', 'suspension', 'maternity', 'study',
                    'compassionate', 'special', 'departmental', 'other_loa'
                 )
            THEN TRUE
            ELSE FALSE
        END AS jamaica_blocks_accrual,

        -- Process flag only: consecutive calendar days >14 (Ex.3 / ESL pack)
        CASE
            WHEN wp.leave_family IN (
                    'vacation', 'sick', 'suspension', 'maternity', 'study',
                    'compassionate', 'special', 'departmental', 'other_loa'
                 )
             AND wp.calendar_days > wp.pause_threshold_days
            THEN TRUE
            ELSE FALSE
        END AS jamaica_pause_applies,

        CASE
            WHEN wp.accrual_per_calendar_day IS NULL THEN NULL
            ELSE ROUND(wp.calendar_days * wp.accrual_per_calendar_day, 4)
        END AS factorial_accrual_during_absence_raw,

        CASE
            WHEN wp.accrual_per_calendar_day IS NULL THEN NULL
            WHEN wp.leave_family IN (
                    'vacation', 'sick', 'suspension', 'maternity', 'study',
                    'compassionate', 'special', 'departmental', 'other_loa'
                 )
            THEN 0.0
            ELSE ROUND(wp.calendar_days * wp.accrual_per_calendar_day, 4)
        END AS jamaica_accrual_during_absence_raw,

        -- Jamaica rounding: >= 0.50 up (floor(x + 0.5))
        CASE
            WHEN wp.accrual_per_calendar_day IS NULL THEN NULL
            ELSE FLOOR(ROUND(wp.calendar_days * wp.accrual_per_calendar_day, 6) + 0.5)
        END AS factorial_accrual_rounded_jamaica_rule,

        CASE
            WHEN wp.eligible_calendar_days_in_year_to_period_end IS NULL
              OR wp.annual_vacation_days IS NULL THEN NULL
            ELSE ROUND(
                (CAST(wp.eligible_calendar_days_in_year_to_period_end AS DOUBLE)
                    / CAST(wp.days_in_year AS DOUBLE))
                * wp.annual_vacation_days
            , 4)
        END AS jamaica_ytd_accrual_if_no_pause_raw,

        CASE
            WHEN wp.eligible_calendar_days_in_year_to_period_end IS NULL
              OR wp.annual_vacation_days IS NULL THEN NULL
            ELSE FLOOR(
                ROUND(
                    (CAST(wp.eligible_calendar_days_in_year_to_period_end AS DOUBLE)
                        / CAST(wp.days_in_year AS DOUBLE))
                    * wp.annual_vacation_days
                , 6) + 0.5
            )
        END AS jamaica_ytd_accrual_rounded,

        CASE
            WHEN wp.previous_return_date IS NOT NULL
             AND wp.start_date > wp.previous_return_date
            THEN date_diff('day', wp.previous_return_date, wp.start_date)
            ELSE NULL
        END AS calendar_days_between_return_and_new_start,

        CASE
            WHEN wp.accrual_per_calendar_day IS NULL THEN NULL
            WHEN wp.previous_return_date IS NOT NULL
             AND wp.start_date > wp.previous_return_date
            THEN ROUND(
                date_diff('day', wp.previous_return_date, wp.start_date)
                * wp.accrual_per_calendar_day
            , 4)
            ELSE NULL
        END AS accrual_between_return_and_new_start,

        'No pause_accrual / stop_accrual field exists on leaves'
            AS schema_evidence_no_pause

    FROM with_previous wp
),

final AS (
    SELECT
        c.*,
        CASE
            WHEN c.factorial_accrual_during_absence_raw IS NULL THEN NULL
            WHEN c.jamaica_blocks_accrual
            THEN c.factorial_accrual_during_absence_raw
            ELSE 0.0
        END AS gap_requires_manual_adjustment_raw,
        CASE
            WHEN c.factorial_accrual_during_absence_raw IS NULL THEN NULL
            WHEN c.jamaica_blocks_accrual
            THEN FLOOR(c.factorial_accrual_during_absence_raw + 0.5)
            ELSE 0
        END AS gap_rounded_jamaica_rule
    FROM calc c
)

SELECT
    employee_code                                       AS "Employee_Code",
    employee_name                                       AS "Employee_Name",
    tenure_start_date                                   AS "Tenure_Start_Date",
    antiquity                                           AS "Antiquity_Factorial",
    contract_version_starts_on                          AS "Contract_Version_Starts_On",
    contract_version_effective_on                       AS "Contract_Version_Effective_On",
    annual_vacation_days                                AS "Annual_Entitlement_Days",
    max_accumulation_days                               AS "Max_Accumulation_3y",
    days_in_year                                        AS "Days_In_Year_365_or_366",
    eligible_calendar_days_in_year_to_period_end        AS "Eligible_Calendar_Days_YTD",
    jamaica_ytd_accrual_if_no_pause_raw                 AS "Jamaica_YTD_Accrual_If_No_Pause",
    jamaica_ytd_accrual_rounded                         AS "Jamaica_YTD_Accrual_Rounded",

    leave_id                                            AS "Leave_ID",
    absence_type                                        AS "Absence_Type",
    leave_family                                        AS "Leave_Family",
    status                                              AS "Status",
    start_date                                          AS "Start_Date",
    end_date                                            AS "End_Date",
    return_date                                         AS "Return_Date",
    duration_days_factorial                             AS "Duration_Days_Factorial_Working",
    calendar_days                                       AS "Calendar_Days",
    pause_threshold_days                                AS "Pause_Threshold_Days",

    CASE
        WHEN jamaica_pause_applies THEN 'YES (>14 calendar days — process / ESL / long vacation flag)'
        WHEN calendar_days > pause_threshold_days
        THEN 'CALENDAR >14 but type outside Jamaica blocking list'
        ELSE 'NO (short absence — still blocks daily accrual if Leave_Family is blocking)'
    END                                                 AS "Absence_Above_Threshold",

    CASE
        WHEN jamaica_blocks_accrual THEN 'YES — Jamaica = 0 on these days; adjust C'
        ELSE 'NO — type not in blocking list'
    END                                                 AS "Blocks_Daily_Accrual",

    ROUND(accrual_per_calendar_day, 6)                  AS "Accrual_Per_Calendar_Day",

    factorial_accrual_during_absence_raw                AS "A_Factorial_Accrual_DURING_Absence",
    jamaica_accrual_during_absence_raw                  AS "B_Jamaica_Accrual_DURING_Absence",
    gap_requires_manual_adjustment_raw                  AS "C_Gap_Requires_Manual_Adjustment",
    factorial_accrual_rounded_jamaica_rule              AS "A_Rounded_Jamaica_Half_Up",
    gap_rounded_jamaica_rule                            AS "C_Gap_Rounded_Jamaica_Half_Up",

    CASE
        WHEN annual_vacation_days IS NULL THEN
            'Cannot calculate. Map this employee to an SIJ vacation tier first.'
        WHEN jamaica_blocks_accrual
         AND COALESCE(gap_requires_manual_adjustment_raw, 0) > 0 THEN
            CONCAT(
                'ACTION: Factorial accrued ~',
                CAST(gap_requires_manual_adjustment_raw AS varchar),
                ' vacation day(s) too much during this leave (',
                absence_type, ', ',
                CAST(calendar_days AS varchar),
                ' calendar days). Manually SUBTRACT ',
                CAST(gap_requires_manual_adjustment_raw AS varchar),
                ' day(s) from this employee''s vacation balance in Factorial ',
                '(Time Off → manual adjustment / incidence). ',
                'Jamaica rule: no accrual on leave days — do this whether the leave is ',
                'above or below 14 days.'
            )
        WHEN jamaica_blocks_accrual THEN
            'ACTION: Blocking leave found, but gap rounds to ~0 for this row. No balance change needed.'
        ELSE
            'No action. This leave type is not in the Jamaica blocking list for vacation accrual.'
    END                                                 AS "What_Client_Should_Do",

    absences_in_period                                  AS "Absences_In_Period",
    previous_return_date                                AS "Previous_Absence_Return_Date",
    calendar_days_between_return_and_new_start          AS "Days_Between_Return_And_New_Start",
    accrual_between_return_and_new_start                AS "D_Accrual_Between_Return_And_New_Start",

    schema_evidence_no_pause                            AS "Schema_Evidence",

    CASE
        WHEN annual_vacation_days IS NULL THEN
            'No SIJ vacation tier mapped for this employee.'
        WHEN jamaica_blocks_accrual THEN
            CONCAT(
                'ADJUST: ', absence_type, ' / ', leave_family,
                ' — ', CAST(calendar_days AS varchar), ' calendar days',
                ' (Factorial working ~', CAST(duration_days_factorial AS varchar), ')',
                '. Tier=', CAST(CAST(annual_vacation_days AS integer) AS varchar),
                '/yr. Factorial accrues ~', CAST(factorial_accrual_during_absence_raw AS varchar),
                '; Jamaica daily rule => B=0; gap C ~',
                CAST(gap_requires_manual_adjustment_raw AS varchar),
                CASE
                    WHEN jamaica_pause_applies THEN ' [also >14 process flag]'
                    ELSE ' [≤14 days — still adjust]'
                END,
                '. No native Factorial stop.'
            )
        WHEN calendar_days > pause_threshold_days THEN
            CONCAT(
                'Long absence (', CAST(calendar_days AS varchar),
                ' calendar days) but type "', absence_type,
                '" is outside Jamaica blocking list — C=0 in this SQL.'
            )
        ELSE
            CONCAT(
                'Non-blocking type. Accrual during absence ~',
                CAST(factorial_accrual_during_absence_raw AS varchar),
                ' (no Jamaica gap for this row).'
            )
    END                                                 AS "Conclusion_Adjustment",

    CASE
        WHEN accrual_between_return_and_new_start IS NOT NULL THEN
            CONCAT(
                'Multiple cycles: ', CAST(calendar_days_between_return_and_new_start AS varchar),
                ' calendar days between return and next start; accrual ~',
                CAST(accrual_between_return_and_new_start AS varchar), '.'
            )
        WHEN absences_in_period > 1 THEN
            'Employee has multiple absences in the period; this is the first in ordered list.'
        ELSE
            'Only 1 absence in the period for this employee.'
    END                                                 AS "Conclusion_Multiple_Cycles_Demonstration"

FROM final
ORDER BY
    calendar_days DESC,
    employee_name,
    start_date;
